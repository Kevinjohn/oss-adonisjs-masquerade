import { test } from '@japa/runner'
import testUtils from '@adonisjs/core/services/test_utils'
import db from '@adonisjs/lucid/services/db'
import User from '#models/user'
import Member from '#models/member'

test.group('fresh host public recipe', (group) => {
  group.setup(() => testUtils.db().migrate())

  test('preserves the second guard and completes the CSRF-protected support journey', async ({
    client,
    assert,
  }) => {
    const hold = Number(process.env.RECIPE_TEST_HOLD_MS ?? 0)
    if (hold > 0 && hold <= 60_000) await new Promise((resolve) => setTimeout(resolve, hold))
    const actor = await User.create({
      email: 'actor@example.test',
      accountIncarnation: 'actor-one',
      securityEpoch: 'actor-epoch-one',
      active: true,
      masqueradeOperator: true,
      canMutate: false,
    })
    const target = await User.create({
      email: 'target@example.test',
      accountIncarnation: 'target-one',
      securityEpoch: 'target-epoch-one',
      active: true,
      masqueradeOperator: false,
      canMutate: true,
    })
    const member = await Member.create({ email: 'member@example.test' })
    const memberResponse = await client.get('/member').withGuard('members').loginAs(member)
    memberResponse.assertStatus(200)
    assert.equal(memberResponse.body().memberId, member.id)

    const idle = await client.get('/masquerade').loginAs(actor)
    idle.assertStatus(200)
    assert.isFalse(idle.body().isMasquerading)
    const idleToken = idle.body().contextToken as string

    const csrfDenied = await client
      .post('/support/start')
      .form({
        targetId: String(target.id),
        reason: 'Recipe check',
        access: 'edit',
        _masqueradeContext: idleToken,
      })
      .loginAs(actor)
    csrfDenied.assertStatus(403)

    const started = await client
      .post('/support/start')
      .redirects(0)
      .withCsrfToken()
      .form({
        targetId: String(target.id),
        reason: 'Recipe check',
        access: 'edit',
        _masqueradeContext: idleToken,
      })
      .loginAs(actor)
    assert.equal(started.status(), 302, 'recipe form start')
    assert.equal(started.header('location'), '/support/customer')
    const activeStatus = await client.get('/masquerade').loginAs(actor)
    activeStatus.assertStatus(200)
    assert.isTrue(activeStatus.body().isMasquerading)
    const activeToken = activeStatus.body().contextToken as string
    assert.notEqual(activeToken, idleToken)

    const active = await client.get('/support/customer').loginAs(actor)
    active.assertStatus(200)
    assert.equal(active.body().actorId, actor.id)
    assert.equal(active.body().effectiveId, target.id)
    assert.equal(active.body().guard, 'web_effective')
    assert.equal(active.body().resolvedActorId, String(actor.id))
    assert.equal(active.body().resolvedEffectiveId, String(target.id))

    const mutated = await client
      .post('/support/customer/preference')
      .withCsrfToken()
      .form({ _masqueradeContext: activeToken })
      .loginAs(actor)
    mutated.assertStatus(200)
    assert.equal(mutated.body().changedId, target.id)
    assert.equal((await db.from('recipe_preferences').first())?.user_id, target.id)

    const blocked = await client
      .post('/support/customer/sensitive')
      .withCsrfToken()
      .form({ _masqueradeContext: activeToken })
      .loginAs(actor)
    blocked.assertStatus(403)
    assert.isUndefined(blocked.body().ran, 'sensitive handler must not execute')

    const stopped = await client
      .post('/support/stop')
      .redirects(0)
      .withCsrfToken()
      .form({ _masqueradeContext: activeToken })
      .loginAs(actor)
    stopped.assertStatus(302)
    assert.equal(stopped.header('location'), '/support')
    const idleStatus = await client.get('/masquerade').loginAs(actor)
    idleStatus.assertStatus(200)
    assert.isFalse(idleStatus.body().isMasquerading)
    assert.notEqual(idleStatus.body().contextToken, activeToken)

    const stale = await client
      .post('/support/customer/preference')
      .withCsrfToken()
      .form({ _masqueradeContext: activeToken })
      .loginAs(actor)
    assert.equal(stale.status(), 409, 'stale context rejection')

    const idleAgain = await client.get('/support/customer').loginAs(actor)
    idleAgain.assertStatus(200)
    assert.equal(idleAgain.body().effectiveId, actor.id)

    const jsonStart = await client
      .post('/masquerade/start')
      .withCsrfToken()
      .form({
        targetId: String(target.id),
        reason: 'JSON control',
        _masqueradeContext: idleStatus.body().contextToken,
      })
      .loginAs(actor)
    jsonStart.assertStatus(200)
    const jsonStop = await client
      .post('/masquerade/stop')
      .withCsrfToken()
      .form({ _masqueradeContext: jsonStart.body().contextToken })
      .loginAs(actor)
    jsonStop.assertStatus(200)

    const bannerStart = await client
      .post('/masquerade/start')
      .withCsrfToken()
      .form({
        targetId: String(target.id),
        reason: 'Generated banner check',
        _masqueradeContext: jsonStop.body().contextToken,
      })
      .loginAs(actor)
    bannerStart.assertStatus(200)
    const bannerPage = await client.get('/support/banner').loginAs(actor)
    bannerPage.assertStatus(200)
    const bannerHtml = bannerPage.text()
    assert.include(bannerHtml, `You are viewing as account ${target.id}.`)
    assert.include(bannerHtml, 'Access: View only.')
    assert.include(bannerHtml, 'action="/masquerade/stop"')
    assert.match(bannerHtml, /name=['"]_csrf['"]/, 'Shield CSRF field')

    const stopForm = /<form\b[^>]*action="([^"]+)"[^>]*>([\s\S]*?)<\/form>/.exec(bannerHtml)
    assert.isNotNull(stopForm, 'generated Stop form')
    if (!stopForm) throw new Error('Generated Stop form was not rendered')
    const inputValue = (name: string) => {
      const input = stopForm?.[2].match(/<input\b[^>]*>/g)?.find((html) =>
        new RegExp(`\\bname=["']${name}["']`).test(html)
      )
      return input?.match(/\bvalue=["']([^"']*)["']/)?.[1]
    }
    const renderedCsrf = inputValue('_csrf')
    const bannerToken = inputValue('_masqueradeContext')
    assert.isString(renderedCsrf, 'Stop form CSRF token')
    if (!renderedCsrf || !bannerToken) throw new Error('Generated Stop form fields are missing')
    assert.equal(bannerToken, bannerStart.body().contextToken)
    const csrfSecret = bannerPage.session('csrf-secret') as string
    assert.isString(csrfSecret, 'Shield session secret')

    const bannerStop = await client
      .post(stopForm[1]!)
      .withSession({ 'csrf-secret': csrfSecret })
      .form({ _csrf: renderedCsrf, _masqueradeContext: bannerToken })
      .loginAs(actor)
    bannerStop.assertStatus(200)
    assert.isFalse(bannerStop.body().isMasquerading)
    assert.notEqual(bannerStop.body().contextToken, bannerToken)
    const idleBanner = await client.get('/support/banner').loginAs(actor)
    idleBanner.assertStatus(204)
    assert.equal(idleBanner.text(), '')

    const publicBannerLayout = await client.get('/support/public-banner')
    publicBannerLayout.assertStatus(204)
    assert.equal(publicBannerLayout.text(), '')
  })
})
